import streamlit as st
from db import run_query, execute_write

def render_customers_page():
    st.header("Customer Accounts Management")
    st.markdown("Register and manage credit/account customers linked to Accounts Receivable.")

    tab1, tab2 = st.tabs(["Registered Customers", "Add New Customer"])

    with tab1:
        st.subheader("Customer Directory")
        customers_df = run_query("SELECT customer_id, account_code, customer_name, phone_no, kra_pin FROM customers ORDER BY customer_name;")
        
        if customers_df.empty:
            st.info("No customer accounts found. Use the 'Add New Customer' tab to register one.")
        else:
            st.dataframe(customers_df, use_container_width=True)

    with tab2:
        st.subheader("Create Customer Account")
        with st.form("new_customer_form"):
            col1, col2 = st.columns(2)
            with col1:
                account_code = st.text_input("Account Code (e.g., CUST-001)")
                customer_name = st.text_input("Customer / Company Name")
            with col2:
                phone_no = st.text_input("Phone Number")
                kra_pin = st.text_input("KRA PIN (Optional)")

            submitted = st.form_submit_button("Save Customer Account")
            if submitted:
                if not account_code or not customer_name:
                    st.error("Account Code and Customer Name are required.")
                else:
                    try:
                        execute_write(
                            """
                            INSERT INTO customers (account_code, customer_name, phone_no, kra_pin)
                            VALUES (:code, :name, :phone, :pin)
                            """,
                            {
                                "code": account_code,
                                "name": customer_name,
                                "phone": phone_no,
                                "pin": kra_pin
                            }
                        )
                        st.success(f"Customer account {customer_name} created successfully!")
                        st.rerun()
                    except Exception as e:
                        st.error(f"Error creating customer (Account Code might already exist): {e}")
